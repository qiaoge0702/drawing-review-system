```python
class TestSWParser:
    """SW解析器测试"""
    
    def test_connect_to_sw(self):
        """测试COM连接"""
        parser = SWParser()
        assert parser.sw_app is not None
        assert parser.sw_app.RevisionNumber.startswith("33.")
    
    def test_parse_lb26_assembly(self):
        """测试LB26装配解析"""
        parser = SWParser()
        assembly = parser.parse_assembly(LB26_ASM_PATH)
        
        assert assembly.name == "LB26.00000拉臂总成"
        assert len(assembly.components) == 45
        
        # 验证标准件识别
        standard_count = sum(1 for c in assembly.components 
                           if "GB/T" in c.name)
        assert standard_count >= 20
    
    def test_bom_accuracy(self):
        """测试BOM准确性"""
        parser = SWParser()
        bom = parser.get_bom(LB26_ASM_PATH)
        
        # 与SW内置BOM对比
        sw_bom = self._get_sw_builtin_bom(LB26_ASM_PATH)
        assert len(bom) == len(sw_bom)
        
        # 字段一致性
        for item, sw_item in zip(bom, sw_bom):
            assert item["name"] == sw_item["name"]
            assert item["quantity"] == sw_item["quantity"]
    
    def test_error_handling(self):
        """测试错误处理"""
        parser = SWParser()
        
        # 文件不存在
        with pytest.raises(RuntimeError):
            parser.parse_assembly("/nonexistent/file.SLDASM")
        
        # 损坏的文件
        with pytest.raises(RuntimeError):
            parser.parse_assembly(CORRUPTED_FILE_PATH)
    
    def test_memory_cleanup(self):
        """测试内存清理"""
        parser = SWParser()
        parser.parse_assembly(LB26_ASM_PATH)
        parser.close_document(LB26_ASM_PATH)
        
        # 验证文档已关闭
        assert parser._get_open_doc(LB26_ASM_PATH) is None
```

```python
class TestGeneratePipeline:
    """生成流水线集成测试"""

        """测试简单零件全流程"""
        task = create_task("simple_bracket.SLDPRT")
        result = run_pipeline(task)
        
        assert result.status == "completed"
        assert result.output_dxf.exists()
        
        # 验证所有步骤完成
        for step in range(1, 8):
            assert result.steps[step].status == "completed"
    
    def test_pipeline_resume(self):
        """测试流水线恢复"""
        task = create_task("test_part.SLDPRT")
        
        # 运行到第4步后中断
        run_pipeline(task, stop_at=4)
        
        # 恢复运行
        result = resume_pipeline(task)
        assert result.status == "completed"
        
        # 前4步不应重新执行
        assert result.steps[3].duration_ms < 100  # 从缓存加载
    
    def test_pipeline_rerun_step(self):
        """测试步骤重跑"""
        task = create_task("test_part.SLDPRT")
        run_pipeline(task)
        
        # 重跑第4步
        result = rerun_step(task, step=4, params={"tolerance_level": "precise"})
        assert result.steps[4].status == "completed"
        
        # 第5-7步应自动重跑
        assert result.steps[5].status == "completed"
        assert result.steps[6].status == "completed"
        assert result.steps[7].status == "completed"
        
        # 第1-3步不应重跑
        assert result.steps[3].execution_count == 1
        assert result.steps[4].execution_count == 2
    
    def test_sw_com_stability(self):
        """测试SW COM稳定性"""
        # 连续运行10个任务
        for i in range(10):
            task = create_task(f"part_{i}.SLDPRT")
            result = run_pipeline(task)
            assert result.status == "completed"
```

```python
class TestDrawingCompliance:

        dxf_path = generate_dxf("test_part.SLDPRT")
        doc = ezdxf.readfile(dxf_path)
        
        dims = list(doc.modelspace().query("DIMENSION"))
        
        # 主要特征应有标注
        assert len(dims) >= 3  # 长、宽、高至少3个

        """测试图层标准"""
        dxf_path = generate_dxf("test_part.SLDPRT")
        doc = ezdxf.readfile(dxf_path)
        
        layer_names = [l.dxf.name for l in doc.layers]
        
        # 必有的图层
        required_layers = ["轮廓线", "标注"]
        for layer in required_layers:
```

```python
class TestPerformance:
    """性能基准测试"""
    
    @pytest.mark.benchmark
    def test_simple_part_generation_time(self):
        """简单零件生成时间"""
        task = create_task("simple_bracket.SLDPRT")
        
        start = time.time()
        result = run_pipeline(task)
        duration = time.time() - start
        
        assert result.status == "completed"
        assert duration < 30  # 秒
    
    @pytest.mark.benchmark
    def test_complex_assembly_generation_time(self):
        """复杂装配生成时间"""
        task = create_task("LB26.11000底架焊合.SLDPRT")
        
        start = time.time()
        result = run_pipeline(task)
        duration = time.time() - start
        
        assert result.status == "completed"
        assert duration < 60  # 秒
    
    @pytest.mark.benchmark
    def test_memory_usage(self):
        """内存使用测试"""
        import psutil
        process = psutil.Process()
        
        mem_before = process.memory_info().rss / 1024 / 1024  # MB
        
        task = create_task("LB26.11000底架焊合.SLDPRT")
        result = run_pipeline(task)
        
        mem_after = process.memory_info().rss / 1024 / 1024  # MB
        mem_peak = mem_after - mem_before
        
        assert result.status == "completed"
        assert mem_peak < 4096  # 4GB
    
    @pytest.mark.benchmark
    def test_concurrent_tasks(self):
        """并发任务测试"""
        tasks = [create_task(f"part_{i}.SLDPRT") for i in range(3)]
        
        start = time.time()
        results = run_concurrent(tasks, max_workers=2)
        duration = time.time() - start
        
        assert all(r.status == "completed" for r in results)
        assert duration < 120  # 2分钟
```
