```python
# app/services/dwg_converter.py
import subprocess
from pathlib import Path
from typing import List, Optional
import logging

logger = logging.getLogger(__name__)


class ConversionResult:
    """转换结果"""
    def __init__(self, success: bool, dxf_path: Optional[Path] = None,
                 message: str = "", entity_retention: float = 0.0):
        self.success = success
        self.dxf_path = dxf_path
        self.message = message
        self.entity_retention = entity_retention


class DWGConverter:
    """
    DWG转DXF转换服务
    
    使用ODA File Converter进行转换
    支持批量转换和质量验证
    """
    
    def __init__(self, oda_path: Optional[str] = None):
        self.oda_path = oda_path or self._find_oda_converter()
        
    def _find_oda_converter(self) -> str:
        """自动查找ODA File Converter"""
        # 常见安装路径
        candidates = [
            "/Applications/ODAFileConverter.app/Contents/MacOS/ODAFileConverter",
            "C:\\Program Files\\ODA\\ODAFileConverter\\ODAFileConverter.exe",
            "/usr/bin/ODAFileConverter",
        ]
        for path in candidates:
            if Path(path).exists():
                return path
        raise FileNotFoundError("ODA File Converter not found")
    
    def convert(self, dwg_path: Path, output_dir: Optional[Path] = None) -> ConversionResult:
        """
        转换单个DWG文件
        
        Args:
            dwg_path: DWG文件路径
            output_dir: 输出目录，默认与DWG同目录
            
        Returns:
            ConversionResult: 转换结果
        """
        dwg_path = Path(dwg_path)
        if not dwg_path.exists():
            return ConversionResult(False, message=f"File not found: {dwg_path}")
        
        output_dir = output_dir or dwg_path.parent
        output_dir.mkdir(parents=True, exist_ok=True)
        
        dxf_path = output_dir / f"{dwg_path.stem}.dxf"
        
        # ODA命令行参数
        # 格式: ODAFileConverter "input" "output" "ACAD2018" "DXF" "0" "1"
        cmd = [
            self.oda_path,
            str(dwg_path.parent),
            str(output_dir),
            "ACAD2018",
            "DXF",
            "0",  # 递归
            "1",  # 覆盖
        ]
        
        try:
            logger.info(f"Converting {dwg_path.name}...")
            result = subprocess.run(
                cmd,
                capture_output=True,
                text=True,
                timeout=60
            )
            
            if result.returncode != 0:
                logger.error(f"Conversion failed: {result.stderr}")
                return ConversionResult(False, message=result.stderr)
            
            if not dxf_path.exists():
                return ConversionResult(False, message="DXF file not created")
            
            # 验证转换质量
            validation = self._validate_conversion(dwg_path, dxf_path)
            
            return ConversionResult(
                success=True,
                dxf_path=dxf_path,
                message="Conversion successful",
                entity_retention=validation.retention_rate
            )
            
        except subprocess.TimeoutExpired:
            return ConversionResult(False, message="Conversion timeout")
        except Exception as e:
            logger.exception("Conversion error")
            return ConversionResult(False, message=str(e))
    
    def _validate_conversion(self, dwg_path: Path, dxf_path: Path) -> 'ValidationReport':
        """验证转换质量"""
        # 使用ezdxf检查DXF完整性
        import ezdxf
        
        try:
            doc = ezdxf.readfile(dxf_path)
            msp = doc.modelspace()
            
            entity_count = len(list(msp))
            layer_count = len(doc.layers)
            
            return ValidationReport(
                is_valid=True,
                entity_count=entity_count,
                layer_count=layer_count,
                retention_rate=1.0  # 简化，实际需要对比DWG
            )
        except Exception as e:
            return ValidationReport(
                is_valid=False,
                error_message=str(e)
            )
    
    def batch_convert(self, dwg_paths: List[Path], 
                      output_dir: Optional[Path] = None,
                      progress_callback=None) -> List[ConversionResult]:
        """
        批量转换
        
        Args:
            dwg_paths: DWG文件路径列表
            output_dir: 统一输出目录
            progress_callback: 进度回调函数(current, total)
            
        Returns:
            List[ConversionResult]: 转换结果列表
        """
        results = []
        total = len(dwg_paths)
        
        for i, dwg_path in enumerate(dwg_paths):
            result = self.convert(dwg_path, output_dir)
            results.append(result)
            
            if progress_callback:
                progress_callback(i + 1, total)
        
        return results


class ValidationReport:
    """转换验证报告"""
    def __init__(self, is_valid: bool, entity_count: int = 0,
                 layer_count: int = 0, retention_rate: float = 0.0,
                 error_message: str = ""):
        self.is_valid = is_valid
        self.entity_count = entity_count
        self.layer_count = layer_count
        self.retention_rate = retention_rate
        self.error_message = error_message
```

```python
# app/renderers/dxf_renderer.py
import matplotlib
matplotlib.use('Agg')  # 无GUI环境
import matplotlib.pyplot as plt
from matplotlib.patches import Circle, Rectangle, Polygon
from matplotlib.lines import Line2D
from PIL import Image
import ezdxf
from pathlib import Path
from typing import Optional, List, Tuple
import numpy as np


class RenderOptions:
    """渲染选项"""
    def __init__(self, 
                 width: int = 2048,
                 height: int = 2048,
                 background_color: str = 'white',
                 line_width_scale: float = 1.0,
                 visible_layers: Optional[List[str]] = None,
                 color_mode: str = 'color'):  # color / monochrome
        self.width = width
        self.height = height
        self.background_color = background_color
        self.line_width_scale = line_width_scale
        self.visible_layers = visible_layers
        self.color_mode = color_mode


class DXFRenderer:
    """
    DXF渲染引擎
    
    将DXF文件渲染为高清PNG图片
    支持图层控制、颜色渲染、文字渲染
    """
    
    # AutoCAD标准颜色表
    COLOR_TABLE = {
        0: (0, 0, 0),
        1: (255, 0, 0),
        2: (255, 255, 0),
        3: (0, 255, 0),
        4: (0, 255, 255),
        5: (0, 0, 255),
        6: (255, 0, 255),
        7: (255, 255, 255),
        8: (128, 128, 128),
        9: (192, 192, 192),
        # ... 更多颜色
    }
    
    def __init__(self):
        self.fig = None
        self.ax = None
    
    def render(self, dxf_path: Path, options: Optional[RenderOptions] = None) -> Image.Image:
        """
        渲染DXF为PNG
        
        Args:
            dxf_path: DXF文件路径
            options: 渲染选项
            
        Returns:
            PIL.Image: 渲染结果
        """
        options = options or RenderOptions()
        
        # 加载DXF
        doc = ezdxf.readfile(dxf_path)
        msp = doc.modelspace()
        
        # 计算图纸范围
        extents = self._calculate_extents(msp)
        
        # 创建图形
        self.fig, self.ax = plt.subplots(figsize=(options.width/100, options.height/100), dpi=100)
        self.ax.set_facecolor(options.background_color)
        
        # 设置坐标范围
        self.ax.set_xlim(extents['min_x'], extents['max_x'])
        self.ax.set_ylim(extents['min_y'], extents['max_y'])
        self.ax.set_aspect('equal')
        self.ax.axis('off')
        
        # 渲染实体
        for entity in msp:
            # 图层过滤
            if options.visible_layers is not None:
                layer = entity.dxf.layer
                if layer not in options.visible_layers:
                    continue
            
            try:
                self._render_entity(entity, options)
            except Exception as e:
                # 忽略单个实体渲染错误
                continue
        
        # 保存为图片
        buf = BytesIO()
        self.fig.savefig(buf, format='png', dpi=100, 
                        bbox_inches='tight', pad_inches=0)
        buf.seek(0)
        img = Image.open(buf)
        
        plt.close(self.fig)
        
        return img
    
    def _calculate_extents(self, msp) -> dict:
        """计算图纸范围"""
        try:
            from ezdxf import bbox
            extents = bbox.extents(msp)
            return {
                'min_x': extents.extmin[0],
                'min_y': extents.extmin[1],
                'max_x': extents.extmax[0],
                'max_y': extents.extmax[1]
            }
        except:
            # 回退：手动计算
            points = []
            for entity in msp:
                try:
                    if hasattr(entity, 'get_points'):
                        points.extend(entity.get_points())
                    elif hasattr(entity.dxf, 'start'):
                        points.append((entity.dxf.start.x, entity.dxf.start.y))
                        points.append((entity.dxf.end.x, entity.dxf.end.y))
                    elif hasattr(entity.dxf, 'center'):
                        points.append((entity.dxf.center.x, entity.dxf.center.y))
                except:
                    continue
            
            if points:
                xs = [p[0] for p in points]
                ys = [p[1] for p in points]
                return {
                    'min_x': min(xs), 'min_y': min(ys),
                    'max_x': max(xs), 'max_y': max(ys)
                }
            
            return {'min_x': 0, 'min_y': 0, 'max_x': 100, 'max_y': 100}
    
    def _render_entity(self, entity, options: RenderOptions):
        """渲染单个实体"""
        entity_type = entity.dxftype()
        
        if entity_type == 'LINE':
            self._render_line(entity, options)
        elif entity_type == 'CIRCLE':
            self._render_circle(entity, options)
        elif entity_type == 'ARC':
            self._render_arc(entity, options)
        elif entity_type == 'LWPOLYLINE':
            self._render_lwpolyline(entity, options)
        elif entity_type == 'TEXT':
            self._render_text(entity, options)
        elif entity_type == 'MTEXT':
            self._render_mtext(entity, options)
        # ... 更多实体类型
    
    def _render_line(self, entity, options: RenderOptions):
        """渲染直线"""
        start = entity.dxf.start
        end = entity.dxf.end
        color = self._get_color(entity, options)
        
        self.ax.plot([start.x, end.x], [start.y, end.y], 
                    color=color, linewidth=1.0 * options.line_width_scale)
    
    def _render_circle(self, entity, options: RenderOptions):
        """渲染圆"""
        center = entity.dxf.center
        radius = entity.dxf.radius
        color = self._get_color(entity, options)
        
        circle = Circle((center.x, center.y), radius, 
                       fill=False, color=color, linewidth=1.0)
        self.ax.add_patch(circle)
    
    def _get_color(self, entity, options: RenderOptions) -> str:
        """获取实体颜色"""
        if options.color_mode == 'monochrome':
            return 'black'
        
        color_index = entity.dxf.color
        if color_index in self.COLOR_TABLE:
            r, g, b = self.COLOR_TABLE[color_index]
            return f'#{r:02x}{g:02x}{b:02x}'
        
        return 'black'
```

```python
# app/services/data_assembler.py
from pathlib import Path
from typing import Optional
from dataclasses import dataclass
from datetime import datetime
import json

from app.parsers.dxf_parser import DXFParser
from app.renderers.dxf_renderer import DXFRenderer, RenderOptions


@dataclass
class DrawingPackage:
    """AI审查数据包"""
    drawing_id: str
    source_file: str
    converted_at: str
    png_path: Path
    json_path: Path
    metadata: dict
    
    def load_image(self) -> bytes:
        """加载图片数据"""
        return self.png_path.read_bytes()
    
    def load_structured_data(self) -> dict:
        """加载结构化数据"""
        return json.loads(self.json_path.read_text())


class DataAssembler:
    """
    数据组装服务
    
    将DWG转换为AI审查所需的完整数据包：
    - PNG图片（AI Vision输入）
    - JSON结构化数据（AI Text输入）
    """
    
    def __init__(self, cache_dir: Path):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.dxf_parser = DXFParser()
        self.dxf_renderer = DXFRenderer()
    
    def assemble(self, dwg_path: Path, 
                 converter: DWGConverter) -> DrawingPackage:
        """
        组装数据包
        
        Args:
            dwg_path: DWG文件路径
            converter: DWG转换器
            
        Returns:
            DrawingPackage: 完整数据包
        """
        import uuid
        
        drawing_id = str(uuid.uuid4())
        work_dir = self.cache_dir / drawing_id
        work_dir.mkdir(exist_ok=True)
        
        # 1. DWG → DXF
        conversion = converter.convert(dwg_path, work_dir)
        if not conversion.success:
            raise ConversionError(conversion.message)
        
        dxf_path = conversion.dxf_path
        
        # 2. DXF → PNG
        img = self.dxf_renderer.render(dxf_path, RenderOptions())
        png_path = work_dir / "render.png"
        img.save(png_path, 'PNG')
        
        # 3. DXF → JSON (复用阶段1 parser)
        drawing = self.dxf_parser.parse(dxf_path)
        structured_data = {
            "drawing_info": drawing.info.model_dump(),
            "metadata": drawing.metadata.model_dump(),
            "extents": drawing.extents.model_dump(),
            "entities": {
                "layer_count": drawing.entities.layer_count,
                "line_count": drawing.entities.line_count,
                "circle_count": drawing.entities.circle_count,
                "arc_count": drawing.entities.arc_count,
                "text_count": drawing.entities.text_count,
                "total_count": drawing.entities.get_total_entity_count()
            }
        }
        
        json_path = work_dir / "data.json"
        json_path.write_text(json.dumps(structured_data, indent=2, default=str))
        
        return DrawingPackage(
            drawing_id=drawing_id,
            source_file=dwg_path.name,
            converted_at=datetime.now().isoformat(),
            png_path=png_path,
            json_path=json_path,
            metadata=structured_data["metadata"]
        )
```
